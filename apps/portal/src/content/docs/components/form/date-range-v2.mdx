import { DocsPage } from "@/components/docs-page";

`DateRangeV2` stores `{ from, to, preset, timezone }`. A recognized preset rebuilds `from` and `to` on refresh. `CUSTOM` keeps the stored bounds. Rolling, calendar, and fixed drafts stay independent, so switching tabs does not convert one into another.

Each box is one `config` field. Omit `modes` for Rolling, Calendar, and Fixed. A non-empty `presets` list also adds Presets in that default. A passed `modes` list controls which tabs show and their order. `presets` in that list with no preset options shows an empty preset select. `timezone: false` hides the timezone row.

## Presets

Omit `modes` and pass `config.presets` to add a Presets mode ahead of Rolling, Calendar, and Fixed. A passed `modes` list decides whether that tab shows. Include `presets` in `modes` without this list to show an empty preset select. Each preset is the date range picker shape: `label`, `value`, `getRange`, and `group` (`recommended`, `relative`, or `calendar`). The select is grouped by `group`. `getRange` receives `{ timezone }` for the zone selected in the picker, and again on reload from `PRESET_ID|TIMEZONE`. A preset that ignores the argument keeps its own dates. Apply stores `value`, that timezone, and the dates from `getRange`. An id in this list wins over a built-in id with the same name.

  contentClassName="place-items-start"
  code={`() => {
    const zoneName = (timezone) =>
      !timezone || timezone === 'LOCALE' ? Intl.DateTimeFormat().resolvedOptions().timeZone : timezone;
    const wallClock = (date, timezone) => {
      const parts = new Intl.DateTimeFormat('en-US', {
        timeZone: zoneName(timezone),
        year: 'numeric',
        month: '2-digit',
        day: '2-digit',
        hour: '2-digit',
        minute: '2-digit',
        second: '2-digit',
        hourCycle: 'h23',
      }).formatToParts(date);
      const value = (type) => Number(parts.find((part) => part.type === type).value);
      return {
        year: value('year'),
        month: value('month'),
        day: value('day'),
        hour: value('hour') % 24,
        minute: value('minute'),
        second: value('second'),
      };
    };
    const daysEarlier = (date, days, timezone) => {
      const wall = wallClock(date, timezone);
      const guess = Date.UTC(wall.year, wall.month - 1, wall.day - days, wall.hour, wall.minute, wall.second);
      const offset = (instant) => {
        const zoned = wallClock(new Date(instant), timezone);
        return Date.UTC(zoned.year, zoned.month - 1, zoned.day, zoned.hour, zoned.minute, zoned.second) - instant;
      };
      const corrected = guess - offset(guess);
      return new Date(guess - offset(corrected));
    };
    const presets = [
      {
        label: 'Previous 3 days',
        value: 'PREVIOUS_3_DAYS',
        group: 'recommended',
        getRange: ({ timezone }) => {
          const to = new Date();
          return { from: daysEarlier(to, 3, timezone), to };
        },
      },
      {
        label: 'Previous 14 days',
        value: 'PREVIOUS_14_DAYS',
        group: 'relative',
        getRange: ({ timezone }) => {
          const to = new Date();
          return { from: daysEarlier(to, 14, timezone), to };
        },
      },
    ];
    const initial = presets[0].getRange({ timezone: 'UTC' });
    const [value, setValue] = React.useState({
      from: initial.from,
      to: initial.to,
      preset: presets[0].value,
      timezone: 'UTC',
    });

    return (
      <Layout.Flex direction="column" gapY="md" align="start">
        <DateRangeV2
          value={value}
          onValueChange={setValue}
          config={{
            direction: 'past',
            calendarProps: { weekStartsOn: 1 },
            tabDetails: [],
            presets,
          }}
        />
        <pre style={{ margin: 0 }}>{JSON.stringify(value, null, 2)}</pre>
      </Layout.Flex>
    );

}`}

## Filter field

`FilterGroup` is the out-of-the-box path. A sticky `FilterFieldTypes.DateRangeV2` option keeps Rolling, Calendar, and Fixed on the built-in query. Presets in `filterFieldConfig.presets` reload through `getRange()`. `calendarProps.weekStartsOn` is used when a calendar week is read back, and it defaults to Monday. Apply and clear update the `timeframe` query param through the filters router. The value below is `location.search`.

## Custom parser

`parser` on the filter option replaces the built-in query for that field. It has to read and write every mode the picker can apply. The parser in this example only accepts `PREVIOUS_3_DAYS` and `PREVIOUS_14_DAYS`.

<DocsPage.ComponentExample
  client:only
  contentClassName="place-items-start"
  code={`() => {
    const location = useLocation();
    const [filters, setFilters] = React.useState({});
    const presets = [
      {
        label: 'Previous 3 days',
        value: 'PREVIOUS_3_DAYS',
        group: 'recommended',
        getRange: () => {
          const to = new Date();
          const from = new Date(to);
          from.setDate(from.getDate() - 3);
          return { from, to };
        },
      },
      {
        label: 'Previous 14 days',
        value: 'PREVIOUS_14_DAYS',
        group: 'relative',
        getRange: () => {
          const to = new Date();
          const from = new Date(to);
          from.setDate(from.getDate() - 14);
          return { from, to };
        },
      },
    ];
    const matchPreset = (id) => presets.find((preset) => preset.value === id);
    const parser = {
      parse: (serialized) => {
        if (!serialized) return undefined;
        const [id, timezone = 'UTC'] = decodeURIComponent(serialized).split('|');
        const preset = matchPreset(id);
        if (!preset) return undefined;
        const range = preset.getRange({ timezone });
        const [from, to] = range.from <= range.to ? [range.from, range.to] : [range.to, range.from];
        return { from, to, preset: preset.value, timezone };
      },
      serialize: (value) => (value && value.preset ? value.preset + '|' + (value.timezone || 'UTC') : ''),
    };
    const query = new URLSearchParams(location.search).get('timeframe');
    const readBack = query ? parser.parse(query) : undefined;

    return (
      <Layout.Flex direction="column" gapY="md" align="start" className="w-full">
        <AlertItem theme="warning" read className="w-full">
          <AlertItem.Description>
            If you pass a custom parser, you need to handle all the modes. Otherwise you may get the following broken experience.
          </AlertItem.Description>
        </AlertItem>
        <FilterGroup
          hideSearch
          hideAddFilter
          handleInputChange={() => {}}
          onFilterValueChange={setFilters}
          filterOptions={[
            {
              type: FilterFieldTypes.DateRangeV2,
              label: 'Timeframe',
              value: 'timeframe',
              sticky: true,
              parser: parser,
              filterFieldConfig: {
                direction: 'any',
                calendarProps: { weekStartsOn: 1 },
                timezone: { browserZone: 'dynamic' },
                tabDetails: [],
                presets: presets,
              },
            },
          ]}
        />
        <Text variant="caption-code">location.search</Text>
        <pre style={{ margin: 0 }}>{location.search || 'empty'}</pre>
        <Text variant="caption-code">parser.parse(timeframe)</Text>
        <pre style={{ margin: 0 }}>{readBack ? JSON.stringify(readBack, null, 2) : 'undefined'}</pre>
      </Layout.Flex>
    );

}`}
